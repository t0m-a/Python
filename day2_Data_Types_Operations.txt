# Data types in Python
# string, integer, bolean. "1234" is a string. 1234 is an integer. Bolean is true or false. 
# "hello"[0] index of the string 0 to 4. This is called subscripting.
print("hello"[4])
# Index can be negative. -1 is the last index, -2 is the second last index and so on.
# It means you start counting from the end of the string.
print("hello"[-1]) # gives the same result than printing "hello"[4]

# float or floating point number = numbers with decimal points. 3.14 is a float. 
# 3 is an integer. 3.0 is a float.

# These are called the major PRIMITIVE data types in Python.
# Convert data types or type casting. You can convert one data type to another data type.
# Converting data type requires a value that is correct for the data type you are converting to. 
# For example, you cannot convert a string that contains letters to an integer.
# int() # Convert to integer data type. It can convert a float to an integer by removing the decimal point.
print(int(3.14)) # Converts the float 3.14 to the integer 3.
# float() # Convert to float data type. It can convert an integer to a float by adding a decimal point.
print(float(3)) # Converts the integer 3 to the float 3.0.
# str() # Convert to string data type. It can convert an integer or a float to a string.
print(str(3.14)) # Converts the float 3.14 to the string "
# bool() # Convert to boolean data type. It can convert an integer or a float to a boolean.
print(bool(0)) # Converts the integer 0 to the boolean False.
print(bool(1)) # Converts the integer 1 to the boolean True.
print(bool("hello")) # Converts the string "hello" to the boolean True.
print(bool("")) # Converts the empty string to the boolean False.
print(bool(3.14)) # Converts the float 3.14 to the boolean True.

# NUMBERS Manipulation.

height = 1.65

print(int(bmi))
print(round(bmi)) # Rounds the float to the nearest integer. 30.8 rounds to 31. Rounds up or down depending on the decimal value.
print(round(bmi, 2)) # Rounds the float to 2 decimal places. round(number, ndigits) where ndigits is the number of decimal places to round to.
