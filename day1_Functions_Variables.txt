# Functions and variables basics
# 
# # print("hello " + input("name ") + "!")
# Careful to nested functions order of processing
name = input("Enter your name: ")
print("Hello " + name + "!")
print(len(name))

# Read functions from right to left, this is a combined function
print(len(input("What is your name?")))
# Using variables to store results of functions
username = input("Enter your username: ")
length = len(username)
print(username)
print(length)

# Swapping variables values
a = 5
b = 4
# reassign values to variables to swap their content
a, b = b, a
print(a)
print(b)
